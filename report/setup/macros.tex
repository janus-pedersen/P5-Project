\NewDocumentCommand{\insertcode}{ O{typescript} m }{
  \StrBehind[2]{#2}{/}[\filename]
  \begin{sourceblock}
    \captionof{source}{\filename}
    \label{code:\filename}
    \inputminted[
      frame=lines,
      framesep=2mm,
      baselinestretch=0.9,
      fontsize=\footnotesize,
      linenos,
      breaklines,
      breakanywhere=true
    ]{#1}{#2}
  \end{sourceblock}
}

\NewDocumentCommand{\insertcodeblock}{ O{typescript} m m m m }{
 \StrBehind[2]{#2}{/}[\filename]
\begin{minipage}{\linewidth}
  \begin{codeblock}
    \captionof{code}{#5 (\textit{\filename} #3--#4)}
    \label{code-block:\filename:#3-#4}
    \inputminted[
      frame=lines,
      framesep=2mm,
      baselinestretch=0.9,
      fontsize=\footnotesize,
      firstline=#3,
      lastline=#4,
      linenos,
      breaklines,
      breakanywhere=true
    ]{#1}{#2}
  \end{codeblock}
  \end{minipage}
}
